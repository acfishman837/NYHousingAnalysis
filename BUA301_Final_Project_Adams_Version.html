compute_auc <- function(model, data, folds, model_name) {
  auc_values <- c()
  
  for (i in 1:length(folds)) {
    # Split data into training and testing sets
    test_index <- folds[[i]]
    train_data <- data[-test_index, ]
    test_data <- data[test_index, ]
    
    # Fit model
    if (model_name == "LPM") {
      trained_model <- lm(denial ~ income, data = train_data)
      predictions <- predict(trained_model, test_data, type = "response")
    } else if (model_name == "logit1") {
      trained_model <- glm(denial ~ income + loan_amount, data = train_data, family = binomial)
      predictions <- predict(trained_model, test_data, type = "response")
    } else if (model_name == "logit2") {
      trained_model <- glm(denial ~ income + loan_amount + property_value + race + sex + occupancy_type + area_income, 
                           data = train_data, family = binomial)
      predictions <- predict(trained_model, test_data, type = "response")
    }
    
    # ROC curve and AUC for the model
    roc_curve <- roc(test_data$denial, predictions)
    auc_values <- c(auc_values, auc(roc_curve))
  }
  
  # Return the average AUC
  mean_auc <- mean(auc_values)
  return(mean_auc)
}
